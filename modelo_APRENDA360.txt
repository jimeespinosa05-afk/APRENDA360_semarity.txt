
CURSOS =
    *id_curso +
    curso +
    exposicion +
    conocimientos

EMPLEADOS =
    *numemp +
    nombre +
    apellidos +
    gerencia

EVENTOS =
    *id_evento +
$CURSOS.id_curso* +
    inicio +
    horas

ASISTENCIAS =
    $EVENTOS.id_evento* +
    $EMPLEADOS.numemp* +
    cdiagnostica +
    cfinal + 
    medida_exposicion<[atendidas] / [horas]> +
    medida_conocimiento<([cfinal] - [cdiagnostica]) / 100> +
    icg<([medida_exposicion] * [exposicion]) +
        ([medida_conocimiento] * [conocimientos])>
