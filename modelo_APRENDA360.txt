

CURSOS =
    *id_curso +
    curso +
    exposicion +
    conocimientos;

EMPLEADOS =
    *numemp +
    nombre +
    apellidos +
    gerencia;

EVENTOS =
    *id_evento +
    $CURSOS.id_curso* +
    inicio +
    horas;

ASISTENCIAS =
    *(
        $EVENTOS.id_evento* +
        $EMPLEADOS.numemp*
    ) +
    atendidas +
    cdiagnostica +
    cfinal +
    medida_exposicion<
        [atendidas] / $EVENTOS.horas
    > +
    medida_conocimiento<
        ([cfinal] - [cdiagnostica]) / 100
    > +
    icg<
        ([medida_exposicion] * $CURSOS.exposicion) +
        ([medida_conocimiento] * $CURSOS.conocimientos)
    >;
